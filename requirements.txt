# Pinned deliberately. Supervisor rebuilds the image on its own schedule, and an
# unpinned `pip install` meant a rebuild could silently pull a breaking FastMCP
# or psycopg release into a running add-on. These are the versions current when
# the working image was built (2026-08), i.e. the closest reproduction of it.
#
# Bump one at a time, rebuild, and check the logs for "schema applied" followed
# by "Uvicorn running" before trusting it.
fastapi==0.141.1
uvicorn==0.52.4
fastmcp==3.4.7
psycopg[binary,pool]==3.3.4

# Streaming JSON parser. Needed so a multi-hundred-MB HAE export can be read
# with constant memory instead of json.loads'ing the whole thing on a Pi.
ijson==3.5.1

# Offline point -> IANA timezone lookup for workout routes (app/localtime.py).
# Ships a ~3 MB abi3 wheel for linux aarch64, no numpy.
tzfpy==2.1.0

# Garmin Connect (app/importers/garmin.py). Unofficial — Garmin has no personal
# API — so pinned hard: a login-flow change upstream is exactly the kind of
# thing an unpinned rebuild would pull in silently. 0.3.20 is the first release
# that survives Garmin's 2026 auth changes (it dropped the dead `garth`) and
# handles the 429 on first login. curl_cffi is its binary transport; pinned too,
# it ships an abi3 aarch64 wheel.
garminconnect==0.3.20
curl_cffi==0.17.0

# FIT decoding. Pure Python, no wheels to worry about.
fitdecode==0.11.0
